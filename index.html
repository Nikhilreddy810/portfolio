@Service
public class BookingService {
    @Transactional(isolation = Isolation.SERIALIZABLE)
    public BookingResponse reserveSeatWithLock(Long flightId, String seatNumber, String idempotencyKey) {
        // 1. Verify Idempotency in Redis
        if (redisTemplate.hasKey("idem:" + idempotencyKey)) {
            return (BookingResponse) redisTemplate.opsForValue().get("idem:" + idempotencyKey);
        }
        // 2. Pessimistic Write Lock on Seat Inventory
        Seat seat = seatRepository.findWithPessimisticLock(flightId, seatNumber)
            .orElseThrow(() -> new SeatUnavailableException("Seat already booked"));
        seat.setStatus(SeatStatus.LOCKED);
        // 3. Persist & cache with TTL
        Booking saved = bookingRepository.save(new Booking(seat));
        redisTemplate.opsForValue().set("idem:" + idempotencyKey, saved, Duration.ofMinutes(15));
        return new BookingResponse("SUCCESS", saved.getId());
    }
}
